// Appended only to the hash-pinned diagnostic copy. No new process/job opens,
// assignments, duplicate handles or changed registry access masks.
function loadMembershipApi() {
  return dlopen('kernel32.dll', {
    IsProcessInJob: {args: [FFIType.ptr, FFIType.ptr, FFIType.ptr], returns: FFIType.i32},
    SetLastError: {args: [FFIType.u32], returns: FFIType.void},
  });
}
let membershipApi: ReturnType<typeof loadMembershipApi> | undefined;
export function ownershipMembership(pid: number) {
  const entry = tracked.get(pid), job = entry && threadJobs.get(entry.threadId);
  if (!entry || !entry.identity || !job || !native) return null;
  const savedError = native.lastError();
  membershipApi ??= loadMembershipApi();
  try {
  const code = new Uint32Array(1);
  if (!native.exitCode(entry.handle, code) || code[0] !== STILL_ACTIVE) return null;
  // Re-read through the exact already-owned handle. A recycled PID cannot
  // satisfy the captured incarnation even if an event arrives late.
  const identity = creationIdentityOf(native, entry.handle);
  if (identity?.incarnation !== entry.identity.incarnation) return null;
  function member(handle: number): number | null {
    if (!handle) return null;
    const value = new Int32Array(1);
    return membershipApi!.symbols.IsProcessInJob(asPointer(entry!.handle), asPointer(handle), ptr(value))
      ? value[0]! : null;
  }
  const proof = {owner: ownershipTrace.owner, pid, created: identity.incarnation, threadId: entry.threadId,
    threadJob: String(job.handle), jobKey: job.key, globalJob: String(globalJob), nestingRefused,
    threadMember: member(job.handle), globalMember: member(globalJob)};
  ownershipTrace.event({event: 'owned-membership', ...proof});
  return proof;
  } finally {membershipApi.symbols.SetLastError(savedError);}
}

// Read only the exact job reference that the registry already owns. This also
// works after its last process handle was closed; no job is opened or duplicated.
export function ownershipJobAccounting(threadId: string) {
  const job = threadJobs.get(threadId);
  if (!job || !native) return null;
  const saved = native.lastError();
  membershipApi ??= loadMembershipApi();
  try {
    const buffer = new Uint8Array(ACCOUNTING_SIZE);
    if (!native.queryJobInfo(job.handle, CLASS_BASIC_AND_IO_ACCOUNTING, buffer)) return null;
    const result = {owner: ownershipTrace.owner, threadId, threadJob: String(job.handle), jobKey: job.key,
      globalJob: String(globalJob), nestingRefused, active: new DataView(buffer.buffer).getUint32(OFF_ACTIVE_PROCESSES, true), wallMs: Date.now()};
    ownershipTrace.event({event: 'owned-job-accounting', ...result}); return result;
  } finally {membershipApi.symbols.SetLastError(saved);}
}
